// URLs do GitHub const urls = { warehouse: 'https://raw.githubusercontent.com/miguelacn1996/logistica_simulador/main/warehouse.glb', rack: 'https://raw.githubusercontent.com/miguelacn1996/logistica_simulador/main/rackpallet.glb', pallet: 'https://raw.githubusercontent.com/miguelacn1996/logistica_simulador/main/palleteuro.glb', forklift: 'https://raw.githubusercontent.com/miguelacn1996/logistica_simulador/main/forklift.glb', worker: 'https://raw.githubusercontent.com/miguelacn1996/logistica_simulador/main/worker.glb' }; let scene, camera, renderer, controls; let worker, forklift, warehouse; let moveVector = new THREE.Vector3(0, 0, 0); let inVehicle = false; const loader = new THREE.GLTFLoader(); init(); function init() { scene = new THREE.Scene(); scene.background = new THREE.Color(0x1a1a1a); camera = new THREE.PerspectiveCamera(60, window.innerWidth / window.innerHeight, 0.1, 1000); camera.position.set(0, 5, 12); renderer = new THREE.WebGLRenderer({ antialias: true }); renderer.setSize(window.innerWidth, window.innerHeight); renderer.shadowMap.enabled = true; document.getElementById('canvas-container').appendChild(renderer.domElement); controls = new THREE.OrbitControls(camera, renderer.domElement); controls.enableDamping = true; controls.dampingFactor = 0.05; controls.maxPolarAngle = Math.PI / 2 - 0.05; // Iluminação forte scene.add(new THREE.AmbientLight(0xffffff, 1.4)); const dirLight = new THREE.DirectionalLight(0xffffff, 1.8); dirLight.position.set(20, 40, 20); dirLight.castShadow = true; scene.add(dirLight); loadAssets(); setupJoystick(); window.addEventListener('resize', onWindowResize); document.getElementById('action-btn').addEventListener('click', toggleVehicle); animate(); } function loadAssets() { let loadedCount = 0; const totalAssets = 5; function checkLoaded() { loadedCount++; const percent = Math.round((loadedCount / totalAssets) * 100); document.getElementById('progress').innerText = percent + '%'; if (loadedCount >= totalAssets) { document.getElementById('loading').style.display = 'none'; } } // 1. Armazém (Aumentado para ser grande e espaçoso) loader.load(urls.warehouse, (gltf) => { warehouse = gltf.scene; warehouse.scale.set(8, 8, 8); // Aumenta o tamanho do hangar warehouse.position.set(0, 0, 0); scene.add(warehouse); checkLoaded(); }, undefined, () => checkLoaded()); // 2. Trabalhador loader.load(urls.worker, (gltf) => { worker = gltf.scene; worker.scale.set(1.2, 1.2, 1.2); worker.position.set(0, 0, 3); scene.add(worker); checkLoaded(); }, undefined, () => { worker = new THREE.Mesh(new THREE.BoxGeometry(0.6, 1.6, 0.4), new THREE.MeshStandardMaterial({color: 0x00ff00})); worker.position.set(0, 0.8, 3); scene.add(worker); checkLoaded(); }); // 3. Empilhador loader.load(urls.forklift, (gltf) => { forklift = gltf.scene; forklift.scale.set(1.2, 1.2, 1.2); forklift.position.set(5, 0, 0); scene.add(forklift); checkLoaded(); }, undefined, () => { forklift = new THREE.Mesh(new THREE.BoxGeometry(1.5, 1.5, 2.5), new THREE.MeshStandardMaterial({color: 0xffcc00})); forklift.position.set(5, 0.75, 0); scene.add(forklift); checkLoaded(); }); // 4. Estantes (Rack) loader.load(urls.rack, (gltf) => { const rack = gltf.scene; rack.scale.set(1.5, 1.5, 1.5); rack.position.set(-8, 0, -6); scene.add(rack); const rack2 = rack.clone(); rack2.position.set(-8, 0, 6); scene.add(rack2); checkLoaded(); }, undefined, () => checkLoaded()); // 5. Palete loader.load(urls.pallet, (gltf) => { const pallet = gltf.scene; pallet.scale.set(1.5, 1.5, 1.5); pallet.position.set(2, 0, 3); scene.add(pallet); checkLoaded(); }, undefined, () => checkLoaded()); } function setupJoystick() { const base = document.getElementById('joystick-base'); const stick = document.getElementById('joystick-stick'); let tracking = false; let center = { x: 0, y: 0 }; const maxDist = 40; base.addEventListener('pointerdown', (e) => { tracking = true; const rect = base.getBoundingClientRect(); center = { x: rect.left + rect.width / 2, y: rect.top + rect.height / 2 }; }); window.addEventListener('pointermove', (e) => { if (!tracking) return; const dx = e.clientX - center.x; const dy = e.clientY - center.y; const dist = Math.min(Math.sqrt(dx * dx + dy * dy), maxDist); const angle = Math.atan2(dy, dx); const constrainedX = Math.cos(angle) * dist; const constrainedY = Math.sin(angle) * dist; stick.style.transform = `translate(${constrainedX}px, ${constrainedY}px)`; moveVector.x = constrainedX / maxDist; moveVector.z = constrainedY / maxDist; }); window.addEventListener('pointerup', () => { if (!tracking) return; tracking = false; stick.style.transform = `translate(0px, 0px)`; moveVector.set(0, 0, 0); }); } function toggleVehicle() { inVehicle = !inVehicle; document.getElementById('mode-text').innerText = inVehicle ? 'A Conduzir Empilhador' : 'A Pé'; document.getElementById('action-btn').innerText = inVehicle ? 'SAIR DO EMPILHADOR' : 'ENTRAR NO EMPILHADOR'; } function updateGame() { const speed = 0.1; const limitX = 18; // Limites das paredes do armazém const limitZ = 18; if (moveVector.lengthSq() > 0) { if (!inVehicle && worker) { let nextX = worker.position.x + moveVector.x * speed; let nextZ = worker.position.z + moveVector.z * speed; // Colisão básica com as paredes para não atravessar if (Math.abs(nextX) < limitX) worker.position.x = nextX; if (Math.abs(nextZ) < limitZ) worker.position.z = nextZ; // Animação de passos (evita deslizar como estátua) worker.position.y = Math.abs(Math.sin(Date.now() * 0.015)) * 0.15; worker.rotation.y = Math.atan2(moveVector.x, moveVector.z); controls.target.copy(worker.position); camera.position.x = worker.position.x; camera.position.z = worker.position.z + 8; } else if (inVehicle && forklift) { let nextX = forklift.position.x + moveVector.x * speed; let nextZ = forklift.position.z + moveVector.z * speed; if (Math.abs(nextX) < limitX) forklift.position.x = nextX; if (Math.abs(nextZ) < limitZ) forklift.position.z = nextZ; forklift.rotation.y = Math.atan2(moveVector.x, moveVector.z); controls.target.copy(forklift.position); camera.position.x = forklift.position.x; camera.position.z = forklift.position.z + 10; } } else if (worker && !inVehicle) { worker.position.y = 0; // Para o balanço quando está parado } // Verificar proximidade ao empilhador if (!inVehicle && worker && forklift) { const dist = worker.position.distanceTo(forklift.position); const btn = document.getElementById('action-btn'); btn.style.display = dist < 3.5 ? 'block' : 'none'; } else if (inVehicle) { document.getElementById('action-btn').style.display = 'block'; } } function onWindowResize() { camera.aspect = window.innerWidth / window.innerHeight; camera.updateProjectionMatrix(); renderer.setSize(window.innerWidth, window.innerHeight); } function animate() { requestAnimationFrame(animate); updateGame(); controls.update(); renderer.render(scene, camera); }
